## 树状数组 Fenwick BIT
* BIT只支持有逆运算的操作
* **如果只维护前缀，不维护区间，则不需要运算的可逆性**

### 半动态区间和（单点修改）
```cpp
template <typename T>
struct BIT
{
    vector<T> c;
    int n;
    BIT(int n) : n(n), c(n + 1, T{}) {} // 树状数组大小为[1,n]
    BIT(vector<T> &a) :n(a.size()-1),c(n+1,0)// 用数组初始化
    {
        vector<T> pre(n + 1, 0);
        for (int i = 1; i <= n; i++)
        {
            pre[i] = pre[i - 1] + a[i];
            c[i] = pre[i] - pre[i - (i & -i)];
        }
    }
    T sum(int x)
    { // 求x位置的前缀和
        T res{};
        for (int i = x; i; i -= i & -i)
            res += c[i];
        return res;
    }
    void add(int x, const T &y)
    { // 在x位置上加上y
        for (int i = x; i <= n; i += i & -i)
            c[i] += y;
    }
    T range(int l, int r)
    { // 计算[l,r]的区间和
        return sum(r) - sum(l - 1);
    }
    int select(const T &k) // 找到最大的下标x,使得sum(x)<=k,维护频次信息时，select(k)+1表示第k大的数，复杂度O(logn) 比在sum上二分O(log^2 n)更快,比对顶堆常数更小
    {
        int x = 0;
        T cur{};
        for (int i = 1 << __lg(n); i; i /= 2)
            if (x + i <= n && cur + c[x + i] <= k)
                x += i, cur += c[x];
        return x;
    }
};
```
### 动态区间和（区间修改）
```cpp
template <typename T>
struct BIT
{
    vector<T> a, b;
    int n;
    BIT(int n) : n(n), a(n + 1, T{}), b(n + 1, T{}) {} // 树状数组大小为[1,n]
    void apply(int l, int r, T val)
    { // 区间修改
        auto add = [&](int pos, T val)
        {
            for (int i = pos; i <= n; i += i & -i)
            {
                a[i] += val;
                b[i] += pos * val;
            }
        };
        add(l, val), add(r + 1, -val);
    }
    i64 query(int l, int r)
    { // 区间和查询
        auto sum = [&](int x)
        {
            T ans = {};
            for (int i = x; i; i -= i & -i)
                ans += (x + 1) * a[i] - b[i];
            return ans;
        };
        return sum(r) - sum(l - 1);
    }
};
```
## ST表（Sparse Table）
可修起始下标
要求RMQ的操作函数具有==区间可加性==，如min，max，gcd，位操作（&，^，|），修改操作函数的时候记得修改ans的初始值
初始化时间复杂度为`O(n*logn)`,查询的复杂度为`O(1)`

```cpp
class ST
{
    vector<vector<int>> f;
    int BG = 0; // 起始下标
    int n;

public:
    ST(const vector<int> &a)
    {
        n = a.size() - BG;
        f.assign(n + 1, vector<int>(int(log2(n)) + 2));
        for (int i = 1; i <= n; i++)
            f[i][0] = a[i - 1 + BG];
        for (int j = 1; (1 << j) <= n; j++)
            for (int i = 1; i + (1 << j) - 1 <= n; i++)
                f[i][j] = max(f[i][j - 1], f[i + (1 << j - 1)][j - 1]); // op
    }
    int RMQ(int l, int r)
    {
        if (l > r) // 非法情况
            return 0;
        l = clamp(l, BG, n), r = clamp(r, BG, n);
        l = l - BG + 1, r = r - BG + 1;
        int L = r - l + 1, ans = -0x3f3f3f3f; // op
        for (int i = 0, j = l; (1 << i) <= L; i++)
        {
            if ((L >> i) & 1)
            {
                ans = max(ans, f[j][i]); // op
                j += (1 << i);
            }
        }
        return ans;
    }
};
```

```cpp
template <typename Info, typename MergeOp>
struct SegTree
{
    int n;
    vector<Info> seg;
    [[no_unique_address]] MergeOp merge; // 空类不占空间

    SegTree(int n_, MergeOp merge_ = {}) : n(n_), merge(move(merge_))
    {
        seg.assign(4 << __lg(n), Info());
        build(1, 1, n);
    }
    SegTree(const vector<Info> &a, MergeOp merge_ = {}) : n(a.size() - 1), merge(move(merge_))
    {
        seg.assign(4 << __lg(n), Info());
        build(1, 1, n, a);
    }

    void pull(int p) { seg[p] = merge(seg[p * 2], seg[p * 2 + 1]); }

    void build(int p, int l, int r, const vector<Info> *a = nullptr)
    {
        if (l == r)
        {
            if (a)
                seg[p] = (*a)[l];
            return;
        }
        int m = (l + r) / 2;
        build(p * 2, l, m, a);
        build(p * 2 + 1, m + 1, r, a);
        pull(p);
    }

    void upd(int p, int l, int r, int pos, const Info &val)
    {
        if (l == r)
        {
            seg[p] = val;
            return;
        }
        int m = (l + r) / 2;
        if (pos <= m)
            upd(p * 2, l, m, pos, val);
        else
            upd(p * 2 + 1, m + 1, r, pos, val);
        pull(p);
    }
    void upd(int pos, const Info &val) { upd(1, 1, n, pos, val); }

    Info qry(int p, int l, int r, int ql, int qr)
    {
        if (ql <= l && r <= qr)
            return seg[p];
        int m = (l + r) / 2;
        if (qr <= m)
            return qry(p * 2, l, m, ql, qr);
        if (ql > m)
            return qry(p * 2 + 1, m + 1, r, ql, qr);
        return merge(qry(p * 2, l, m, ql, qr), qry(p * 2 + 1, m + 1, r,
                                                   ql, qr));
    }
    Info qry(int l, int r) { return qry(1, 1, n, l, r); }
};
```

### WIDA模板线段树

```cpp
template<class T> struct Segt {
    struct node {
        int l, r;
        T w, rmq, lazy;
    };
    vector<T> w;
    vector<node> t;

    Segt() {}
    Segt(int n) { init(n); }
    Segt(vector<int> in) {
        int n = in.size() - 1;
        w.resize(n + 1);
        for (int i = 1; i <= n; i++) {
            w[i] = in[i];
        }
        init(in.size() - 1);
    }
    
    #define GL (k << 1)
    #define GR (k << 1 | 1)
    
    void init(int n) {
        w.resize(n + 1);
        t.resize(n * 4 + 1);
        auto build = [&](auto self, int l, int r, int k = 1) {
            if (l == r) {
                t[k] = {l, r, w[l], w[l], -1}; // 如果有赋值为 0 的操作，则懒标记必须要 -1
                return;
            }
            t[k] = {l, r, 0, 0, -1};
            int mid = (l + r) / 2;
            self(self, l, mid, GL);
            self(self, mid + 1, r, GR);
            pushup(k);
        };
        build(build, 1, n);
    }
    void pushdown(node &p, T lazy) { //make_tag
    /* 【在此更新下递函数】 */
        p.w += (p.r - p.l + 1) * lazy;
        p.rmq += lazy;
        if (p.lazy == -1) p.lazy = 1;
	    p.lazy += lazy;
    }
    void pushdown(int k) {
        if (t[k].lazy == -1) return;
        pushdown(t[GL], t[k].lazy);
        pushdown(t[GR], t[k].lazy);
        t[k].lazy = -1;
    }
    void pushup(int k) {
        auto pushup = [&](node &p, node &l, node &r) { 
        /* 【在此更新上传函数】 */
            p.w = l.w + r.w;
            p.rmq = min(l.rmq, r.rmq); // RMQ -> min/max
        };
        pushup(t[k], t[GL], t[GR]);
    }
    void modify(int l, int r, T val, int k = 1) { // 区间修改
        if (l <= t[k].l && t[k].r <= r) {
            pushdown(t[k], val);
            return;
        }
        pushdown(k);
        int mid = (t[k].l + t[k].r) / 2;
        if (l <= mid) modify(l, r, val, GL);
        if (mid < r) modify(l, r, val, GR);
        pushup(k);
    }
    T rmq(int l, int r, int k = 1) { // 区间询问最小值
        if (l <= t[k].l && t[k].r <= r) {
            return t[k].rmq;
        }
        pushdown(k);
        int mid = (t[k].l + t[k].r) / 2;
        T ans = numeric_limits<T>::max(); // RMQ -> 为 max 时需要修改为 ::lowest()
        if (l <= mid) ans = min(ans, rmq(l, r, GL)); // RMQ -> min/max
        if (mid < r) ans = min(ans, rmq(l, r, GR)); // RMQ -> min/max
        return ans;
    }
    T ask(int l, int r, int k = 1) { // 区间询问
        if (l <= t[k].l && t[k].r <= r) {
            return t[k].w;
        }
        pushdown(k);
        int mid = (t[k].l + t[k].r) / 2;
        T ans = 0;
        if (l <= mid) ans += ask(l, r, GL);
        if (mid < r) ans += ask(l, r, GR);
        return ans;
    }
    void debug(int k = 1) {
        cout << "[" << t[k].l << ", " << t[k].r << "]: ";
        cout << "w = " << t[k].w << ", ";
        cout << "Min = " << t[k].rmq << ", ";
        cout << "lazy = " << t[k].lazy << ", ";
        cout << endl;
        if (t[k].l == t[k].r) return;
        debug(GL), debug(GR);
    }
};
```

### 全功能线段树 assign rmq sum mul 
```cpp
template <class T>
struct SegTree
{
    struct Node
    {
        int l, r;
        T sum = 0, rmq; // info
        T add = 0, mul = 1; // lazy
        T assign;
        bool has_assign;
    };

    vector<T> a;
    vector<Node> t;

    SegTree() {}
    SegTree(int n) : a(n + 1, 0) { init(n); }
    SegTree(const vector<T> &in)
    {
        int n = in.size() - 1;
        a = in;
        init(n);
    }

    void init(int n)
    {
        t.resize(4 * n + 5);
        build(1, 1, n);
    }

private:
    void apply_assign(Node &p, T val)
    {
        int len = p.r - p.l + 1;
        p.sum = val * len;
        p.rmq = val;
        p.assign = val;
        p.has_assign = true;
        p.add = Node().add;
        p.mul = Node().mul;
    }

    void apply_add(Node &p, T val)
    {
        int len = p.r - p.l + 1;
        p.sum += val * len;
        p.rmq += val;
        if (p.has_assign)
            p.assign += val;
        else
            p.add += val;
    }

    void apply_mul(Node &p, T val)
    {
        p.sum *= val;
        p.rmq *= val;
        if (p.has_assign)
            p.assign *= val;
        else
        {
            p.mul *= val;
            p.add *= val;
        }
    }

    void pushdown(int k)
    {
        auto &p = t[k];
        if (p.has_assign)
        {
            apply_assign(t[k << 1], p.assign);
            apply_assign(t[k << 1 | 1], p.assign);
            p.has_assign = false;
        }
        if (p.mul != 1)
        {
            apply_mul(t[k << 1], p.mul);
            apply_mul(t[k << 1 | 1], p.mul);
            p.mul = 1;
        }
        if (p.add != 0)
        {
            apply_add(t[k << 1], p.add);
            apply_add(t[k << 1 | 1], p.add);
            p.add = 0;
        }
    }

    void pushup(int k)
    {
        t[k].sum = t[k << 1].sum + t[k << 1 | 1].sum;
        t[k].rmq = rmq_op(t[k << 1].rmq, t[k << 1 | 1].rmq);
    }

    void build(int k, int l, int r)
    {
        t[k].l = l;
        t[k].r = r;
        t[k].add = 0;
        t[k].mul = 1;
        t[k].has_assign = false;
        if (l == r)
        {
            t[k].sum = t[k].rmq = a[l];
            return;
        }
        int m = (l + r) / 2;
        build(k << 1, l, m);
        build(k << 1 | 1, m + 1, r);
        pushup(k);
    }

public:
    void assign(int l, int r, T val, int k = 1)
    {
        if (l <= t[k].l && t[k].r <= r)
        {
            apply_assign(t[k], val);
            return;
        }
        pushdown(k);
        int m = (t[k].l + t[k].r) / 2;
        if (l <= m)
            assign(l, r, val, k << 1);
        if (r > m)
            assign(l, r, val, k << 1 | 1);
        pushup(k);
    }

    void add(int l, int r, T val, int k = 1)
    {
        if (l <= t[k].l && t[k].r <= r)
        {
            apply_add(t[k], val);
            return;
        }
        pushdown(k);
        int m = (t[k].l + t[k].r) / 2;
        if (l <= m)
            add(l, r, val, k << 1);
        if (r > m)
            add(l, r, val, k << 1 | 1);
        pushup(k);
    }

    void mul(int l, int r, T val, int k = 1)
    {
        if (l <= t[k].l && t[k].r <= r)
        {
            apply_mul(t[k], val);
            return;
        }
        pushdown(k);
        int m = (t[k].l + t[k].r) / 2;
        if (l <= m)
            mul(l, r, val, k << 1);
        if (r > m)
            mul(l, r, val, k << 1 | 1);
        pushup(k);
    }

    T range_sum(int l, int r, int k = 1)
    {
        if (l <= t[k].l && t[k].r <= r)
            return t[k].sum;
        pushdown(k);
        int m = (t[k].l + t[k].r) / 2;
        T res = 0;
        if (l <= m)
            res += range_sum(l, r, k << 1);
        if (r > m)
            res += range_sum(l, r, k << 1 | 1);
        return res;
    }

    T rmq(int l, int r, int k = 1) // 区间rmq查询，可自定义
    {
        /*
            自定义rmq需要修改res初始值与rmq_op合并运算
        */
        if (l <= t[k].l && t[k].r <= r)
            return t[k].rmq;
        pushdown(k);
        int m = (t[k].l + t[k].r) / 2;
        T res = -1e18; // RMQ <---- 初始值
        if (l <= m)
            res = rmq_op(res, rmq(l, r, k << 1));
        if (r > m)
            res = rmq_op(res, rmq(l, r, k << 1 | 1));
        return res;
    }

    T rmq_op(T x, T y) { return max(x, y); } // RMQ <---- rmq运算函数
};
```
### Info线段树（min,max,sum)
```cpp
// 自定义标记结构
struct tag
{
    static LL MOD;
    LL add = 0;
    LL mul = 1;
    LL assign = 0;
    bool has_assign = false;

    tag() : add(0), mul(1), assign(0), has_assign(false) {}
    tag(LL a) : add(a), mul(1), assign(0), has_assign(false) {}
    tag(LL a, LL m, LL as, bool h) : add(a), mul(m), assign(as), has_assign(h) {}
};
LL tag::MOD = 998244353;
// 自定义信息结构
struct info
{
    LL sum = 0;
    LL rmq = 0;
    int len = 0;

    info() {}
    info(LL x) : sum(x),rmq(x), len(1) {}

    // 应用标记
    info apply(tag t, int new_len) const
    {
        info res = *this;
        if (t.has_assign)
        {
            res.sum = t.assign * new_len;
            res.rmq = t.assign;
            return res;
        }
        if (t.mul != 1)
        {
            res.sum *= t.mul;
            res.rmq *= t.mul;
            // if (t.mul < 0)
            //     swap(res.min_val, res.max_val);
        }
        if (t.add != 0)
        {
            res.sum += t.add * new_len;
            res.rmq += t.add;
        }
        return res;
    }
};
// 标记合并（先a后b）
tag operator+(const tag &a, const tag &b)
{
    if (b.has_assign)
        return b;
    tag res = a;
    if (b.mul != 1)
    {
        if (res.has_assign)
            res.assign *= b.mul;
        else
            res.mul *= b.mul;
        res.add *= b.mul;
    }
    if (b.add != 0)
    {
        if (res.has_assign)
            res.assign += b.add;
        else
            res.add += b.add;
    }
    return res;
}
// 区间合并
info operator+(const info &a, const info &b)
{
    if (a.len == 0)
        return b;
    if (b.len == 0)
        return a;
    info res;
    res.sum = a.sum + b.sum;
    res.rmq = max(a.rmq, b.rmq);// <------rmq
    res.len = a.len + b.len;
    return res;
}

class stree
{
    vector<info> w;
    vector<tag> lazy;
    int n;

    void maketag(int u, int len, const tag &t)
    {
        lazy[u] = lazy[u] + t;
        w[u] = w[u].apply(t, len);
    }

    void pushdown(int u, int L, int R)
    {
        if (!lazy[u].has_assign && lazy[u].mul == 1 && lazy[u].add == 0)
            return;
        int M = (L + R) >> 1;
        maketag(u << 1, M - L + 1, lazy[u]);
        maketag(u << 1 | 1, R - M, lazy[u]);
        lazy[u] = tag();
    }

    void pushup(int u)
    {
        w[u] = w[u << 1] + w[u << 1 | 1];
    }

    info query(int u, int L, int R, int l, int r)
    {
        if (l <= L && R <= r)
        {
            return w[u];
        }
        else if (L <= r && R >= l)
        {
            int M = (L + R) >> 1;
            pushdown(u, L, R);
            return query(u << 1, L, M, l, r) + query(u << 1 | 1, M + 1, R, l, r);
        }
        else
        {
            return info();
        }
    }

    void insert(int u, int L, int R, int p, const info &x)
    {
        if (L == R)
        {
            w[u] = x;
            lazy[u] = tag();
        }
        else
        {
            pushdown(u, L, R);
            int M = (L + R) >> 1;
            if (p <= M)
                insert(u << 1, L, M, p, x);
            else
                insert(u << 1 | 1, M + 1, R, p, x);
            pushup(u);
        }
    }

    void build(int u, int L, int R, const vector<int> &a)
    {
        if (L == R)
        {
            w[u] = info(a[L]);
            lazy[u] = tag();
            return;
        }
        int M = (L + R) >> 1;
        build(u << 1, L, M, a);
        build(u << 1 | 1, M + 1, R, a);
        pushup(u);
    }

    void update(int u, int L, int R, int l, int r, const tag &x)
    {
        if (l <= L && R <= r)
        {
            maketag(u, R - L + 1, x);
        }
        else if (L <= r && R >= l)
        {
            int M = (L + R) >> 1;
            pushdown(u, L, R);
            update(u << 1, L, M, l, r, x);
            update(u << 1 | 1, M + 1, R, l, r, x);
            pushup(u);
        }
    }

public:
    stree(const vector<int> &a)
    {
        n = a.size() - 1;
        w.resize(4 * n + 5);
        lazy.resize(4 * n + 5);
        if (n)
            build(1, 1, n, a);
    }

    info query(int p) { return query(1, 1, n, p, p); }
    info query(int l, int r) { return query(1, 1, n, l, r); }
    void insert(int p, const info &x) { insert(1, 1, n, p, x); }
    void range_add(int l, int r, LL v) { update(1, 1, n, l, r, tag(v)); }
```

## 珂朵莉树（OD Tree)

区间赋值的数据结构都可以骗分，在数据随机的情况下，复杂度可以保证，时间复杂度：$\mathcal O(N\log\log N)$ 。

```c++
struct ODT // 珂朵莉树 [1, n]
{
    struct Node
    {
        int l, r;
        mutable int v;
        Node(int l, int r = -1, int v = 0) : l(l), r(r), v(v) {}
        bool operator<(const Node &o) const { return l < o.l; }
    };
    set<Node> s;
    ODT() { s.clear(); }
    ODT(int n)
    {
        s.clear();
        s.emplace(1, n, 0);
    }

    set<Node>::iterator split(int pos)
    {
        auto it = s.lower_bound({pos});
        if (it != s.end() && it->l == pos)
            return it;
        it--;
        auto [l, r, v] = *it;
        s.erase(it);
        s.emplace(l, pos - 1, v);
        return s.emplace(pos, r, v).first;
    }
    void assign(int l, int r, int x)
    {
        auto itr = split(r + 1), itl = split(l);
        s.erase(itl, itr);
        s.emplace(l, r, x);
    }
    void add(int l, int r, int x)
    {
        auto itr = split(r + 1), itl = split(l);
        for (auto it = itl; it != itr; it++)
            it->v += x;
    }
    int kth(int l, int r, int k)
    {
        vector<pair<int, int>> a;
        auto itr = split(r + 1), itl = split(l);
        for (auto it = itl; it != itr; it++)
            a.push_back({it->v, it->r - it->l + 1});
        sort(ALL(a));
        for (auto [val, len] : a)
        {
            k -= len;
            if (k <= 0)
                return val;
        }
        return -1;
    }
    int powersum(int l, int r, int x, int mod)
    {
        auto itr = split(r + 1), itl = split(l);
        int ans = 0;
        for (auto it = itl; it != itr; it++)
        {
            int len = it->r - it->l + 1;
            int val = it->v % mod;
            int pow = 1;
            for (int b = x; b; b >>= 1, val = val * val % mod)
                if (b & 1)
                    pow = pow * val % mod;
            ans = (ans + pow * len) % mod;
        }
        return ans;
    }
};
```

```cpp
template <typename T, typename F, int NODES = 20000000>
struct PersistentSegmentTree
{
    using ll = long long;
    struct Node
    {
        T data;
        Node *l, *r;
        Node() {}
        Node(const T &_data) : data(_data), l(nullptr), r(nullptr) {}
    };

    Node *pool;
    int pid;
    ll N;
    const F f;
    const T ID;
    Node *nil;
    vector<Node *> roots;

    PersistentSegmentTree(const vector<T> &v, const F &f_, const T &ID_)
        : pid(0), f(f_), ID(ID_), nil(nullptr)
    {
        pool = new Node[NODES];
        nil = my_new(ID);
        nil->l = nil->r = nil;
        roots.reserve(262144);
        roots.push_back(build(v));
    }

    PersistentSegmentTree(const ll &N_, const F &f_, const T &ID_)
        : pid(0), N(N_), f(f_), ID(ID_), nil(nullptr)
    {
        pool = new Node[NODES];
        nil = my_new(ID);
        nil->l = nil->r = nil;
        roots.reserve(262144);
        roots.push_back(nil);
    }

    Node *my_new(const T &data)
    {
        pool[pid].data = data;
        pool[pid].l = pool[pid].r = nil;
        return &(pool[pid++]);
    }

    Node *merge(Node *l, Node *r)
    {
        pool[pid].data = f(l->data, r->data);
        pool[pid].l = l;
        pool[pid].r = r;
        return &(pool[pid++]);
    }

    Node *build(const vector<T> &v)
    {
        N = (ll)v.size();
        return build(0, (ll)v.size(), v);
    }

    Node *build(ll l, ll r, const vector<T> &v)
    {
        if (l + 1 == r)
            return my_new(v[l]);
        ll m = (l + r) >> 1;
        return merge(build(l, m, v), build(m, r, v));
    }

private:
    Node *update_(ll a, const T &x, Node *n, ll l, ll r)
    {
        if (l + 1 == r)
            return my_new(x);
        ll m = (l + r) >> 1;
        if (a < m)
            return merge(update_(a, x, n->l, l, m), n->r);
        return merge(n->l, update_(a, x, n->r, m, r));
    }
    Node *add_(ll a, const T &x, Node *n, ll l, ll r)
    {
        if (l + 1 == r)
            return my_new(f(x, n->data));
        ll m = (l + r) >> 1;
        if (a < m)
            return merge(add_(a, x, n->l, l, m), n->r);
        return merge(n->l, add_(a, x, n->r, m, r));
    }
    T query_(ll a, ll b, Node *n, ll l, ll r)
    {
        if (n == nil)
            return ID;
        if (r <= a or b <= l)
            return ID;
        if (a <= l and r <= b)
            return n->data;
        ll m = (l + r) >> 1;
        return f(query_(a, b, n->l, l, m), query_(a, b, n->r, m, r));
    }

public:
    Node *update(Node *n, ll k, const T &x)
    {
        Node *root = update_(k, x, n, 0, N);
        roots.push_back(root);
        return root;
    }
    Node *update(int t, ll k, const T &x)
    {
        Node *root = update_(k, x, roots[t], 0, N);
        roots.push_back(root);
        return root;
    }
    Node *update(ll k, const T &x)
    {
        Node *root = update_(k, x, roots.back(), 0, N);
        roots.push_back(root);
        return root;
    }

    Node *add(Node *n, ll k, const T &x)
    {
        Node *root = add_(k, x, n, 0, N);
        roots.push_back(root);
        return root;
    }
    Node *add(int t, ll k, const T &x)
    {
        Node *root = add_(k, x, roots[t], 0, N);
        roots.push_back(root);
        return root;
    }
    Node *add(ll k, const T &x)
    {
        Node *root = add_(k, x, roots.back(), 0, N);
        roots.push_back(root);
        return root;
    }

    T query(Node *n, ll a, ll b) { return query_(a, b, n, 0, N); }
    T query(int t, ll a, ll b) { return query_(a, b, roots[t], 0, N); }
    T query(ll a, ll b) { return query_(a, b, roots.back(), 0, N); }

    Node *new_tree() { return nil; }
};
```

```cpp
struct PresidentTree {
    static constexpr int N = 2e5 + 10;
    int cntNodes, root[N];
    struct node {
        int l, r;
        int cnt;
    } tr[4 * N + 17 * N];
    void modify(int &u, int v, int l, int r, int x) {
        u = ++cntNodes;
        tr[u] = tr[v];
        tr[u].cnt++;
        if (l == r) return;
        int mid = (l + r) / 2;
        if (x <= mid)
            modify(tr[u].l, tr[v].l, l, mid, x);
        else
            modify(tr[u].r, tr[v].r, mid + 1, r, x);
    }
    int kth(int u, int v, int l, int r, int k) {
        if (l == r) return l;
        int res = tr[tr[v].l].cnt - tr[tr[u].l].cnt;
        int mid = (l + r) / 2;
        if (k <= res)
            return kth(tr[u].l, tr[v].l, l, mid, k);
        else
            return kth(tr[u].r, tr[v].r, mid + 1, r, k - res);
    }
};
```

```cpp
struct Tire
{
    vector<vector<int>> ch;
    vector<int> cnt;
    int idx;
    map<char, int> mp;
    Tire(int n = 1000)
    {
        ch.assign(n, vector<int>(63));
        cnt.assign(n, 0);
        idx = 0;
        LL id = 0;
        for (char c = 'a'; c <= 'z'; c++)
            mp[c] = ++id;
        for (char c = 'A'; c <= 'Z'; c++)
            mp[c] = ++id;
        for (char c = '0'; c <= '9'; c++)
            mp[c] = ++id;
    }
    void insert(const string &s)
    {
        int u = 0;
        for (int i = 0; i < s.size(); i++)
        {
            int id = mp[s[i]];
            if (!ch[u][id])
                ch[u][id] = ++idx;
            u = ch[u][id];
        }
        cnt[u]++;
    }
    LL query(const string &s)
    {
        int u = 0;
        for (int i = 0; i < s.size(); i++)
        {
            int id = mp[s[i]];
            if (!ch[u][id])
                return 0;
            u = ch[u][id];
        }
        return cnt[u];
    }
    void clear()
    {
        for (int i = 0; i <= idx; i++)
        {
            cnt[i] = 0;
            for (int j = 0; j <= 62; j++)
                ch[i][j] = 0;
        }
        idx = 0;
    }
};
```

```cpp
// Trie+Kmp，多模式串匹配
struct ACAutomaton {
    static constexpr int N = 1e6 + 10;
    int ch[N][26], fail[N], cntNodes;
    int cnt[N];
    ACAutomaton() {
        cntNodes = 1;
    }
    void insert(string s) {
        int u = 1;
        for (auto c : s) {
            int &v = ch[u][c - 'a'];
            if (!v) v = ++cntNodes;
            u = v;
        }
        cnt[u]++;
    }
    void build() {
        fill(ch[0], ch[0] + 26, 1);
        queue<int> q;
        q.push(1);
        while (!q.empty()) {
            int u = q.front();
            q.pop();
            for (int i = 0; i < 26; i++) {
                int &v = ch[u][i];
                if (!v)
                    v = ch[fail[u]][i];
                else {
                    fail[v] = ch[fail[u]][i];
                    q.push(v);
                }
            }
        }
    }
    LL query(string t) {
        LL ans = 0;
        int u = 1;
        for (auto c : t) {
            u = ch[u][c - 'a'];
            for (int v = u; v && ~cnt[v]; v = fail[v]) {
                ans += cnt[v];
                cnt[v] = -1;
            }
        }
        return ans;
    }
};
```
