using i64 = long long;

template<class T> constexpr T mypow(T n, i64 k) {
    T r = 1;
    for (; k; k /= 2, n *= n) {
        if (k % 2) {
            r *= n;
        }
    }
    return r;
}

template<class T> constexpr T power(int n) {
    return mypow(T(2), n);
}

template<const int &MOD> struct Zmod {
    int x;
    Zmod(signed x = 0) : x(norm(x % MOD)) {}
    Zmod(i64 x) : x(norm(x % MOD)) {}

    constexpr int norm(int x) const noexcept {
        if (x < 0) [[unlikely]] {
            x += MOD;
        }
        if (x >= MOD) [[unlikely]] {
            x -= MOD;
        }
        return x;
    }
    explicit operator int() const {
        return x;
    }
    constexpr int val() const {
        return x;
    }
    constexpr Zmod operator-() const {
        Zmod val = norm(MOD - x);
        return val;
    }
    constexpr Zmod inv() const {
        assert(x != 0);
        return mypow(*this, MOD - 2);
    }
    friend constexpr auto &operator>>(istream &in, Zmod &j) {
        int v;
        in >> v;
        j = Zmod(v);
        return in;
    }
    friend constexpr auto &operator<<(ostream &o, const Zmod &j) {
        return o << j.val();
    }
    constexpr Zmod &operator++() {
        x = norm(x + 1);
        return *this;
    }
    constexpr Zmod &operator--() {
        x = norm(x - 1);
        return *this;
    }
    constexpr Zmod operator++(signed) {
        Zmod res = *this;
        ++*this;
        return res;
    }
    constexpr Zmod operator--(signed) {
        Zmod res = *this;
        --*this;
        return res;
    }
    constexpr Zmod &operator+=(const Zmod &i) {
        x = norm(x + i.x);
        return *this;
    }
    constexpr Zmod &operator-=(const Zmod &i) {
        x = norm(x - i.x);
        return *this;
    }
    constexpr Zmod &operator*=(const Zmod &i) {
        x = i64(x) * i.x % MOD;
        return *this;
    }
    constexpr Zmod &operator/=(const Zmod &i) {
        return *this *= i.inv();
    }
    constexpr Zmod &operator%=(const int &i) {
        return x %= i, *this;
    }
    friend constexpr Zmod operator+(const Zmod i, const Zmod j) {
        return Zmod(i) += j;
    }
    friend constexpr Zmod operator-(const Zmod i, const Zmod j) {
        return Zmod(i) -= j;
    }
    friend constexpr Zmod operator*(const Zmod i, const Zmod j) {
        return Zmod(i) *= j;
    }
    friend constexpr Zmod operator/(const Zmod i, const Zmod j) {
        return Zmod(i) /= j;
    }
    friend constexpr Zmod operator%(const Zmod i, const int j) {
        return Zmod(i) %= j;
    }
    friend constexpr bool operator==(const Zmod i, const Zmod j) {
        return i.val() == j.val();
    }
    friend constexpr bool operator!=(const Zmod i, const Zmod j) {
        return i.val() != j.val();
    }
    friend constexpr bool operator<(const Zmod i, const Zmod j) {
        return i.val() < j.val();
    }
    friend constexpr bool operator>(const Zmod i, const Zmod j) {
        return i.val() > j.val();
    }
};

## 正则括号序列（RBS）

*定义：$( \to +1,\ ) \to -1$，合法序列 $\iff$ 全程 balance $\ge 0$ 且最终 $=0$。*

- **最大合法子串长度**：栈存下标，遇到匹配的 `)` 时弹栈，区间 `[stk.top()+1, i]` 为合法子串
- **最少添加括号数**：遍历中 balance 最低到达 $-k$，末尾 balance 为 $b$，则需要补 $k$ 个 `(` 和 $\max(0, b+k)$ 个 `)`
- **嵌套深度**：遍历时 balance 的最大值即最大嵌套层数
- **交换相邻括号**：`)(` $\to$ `()` 会使 balance $+2$，这是最少交换次数类问题的核心观察
- **字典序下一个合法序列**：从后往前找到第一个能改成 `(` 的位置（需满足 balance 约束），改成 `(` 后尽量填 `(`
- **本质不同合法括号序列计数**：$n$ 对括号 $\Rightarrow$ Catalan 数 $C_n = \frac{1}{n+1}\binom{2n}{n}$

### RBS建树
对一棵有根树做 DFS，进入节点时写一个`(`，离开节点时写一个`)`，得到的序列就是合法括号序列；反之，任何合法括号序列都可以还原为一棵有序有根树。
```cpp
auto RBSTREE = [&](string s)
{
	vector<vector<int>> g(s.size() / 2 + 1);
	stack<int> stk;
	stk.emplace(0); // 虚拟根节点0
	int cur = 0, cnt = 0;
	for (char c : s)
	{
		if (c == '(')
		{
			cur++;
			if (!stk.empty()) // 如果有父节点
			{
				int fa = stk.top(); // 父节点是上一级括号
				g[fa].emplace_back(cur);
			}
			stk.emplace(cur);
		}
		else
			stk.pop();
	}
	return g;
};
```

class Mex
{
    map<int, int> cnt;
    set<int> missing;
    int size;

public:
    Mex(int _size = 100) : size(_size), missing(from_range, views::iota(0, _size + 1)) {};
    void insert(int x)
    {
        cnt[x]++;
        if (cnt[x] == 1)
            missing.erase(x);
    }

    void erase(int x)
    {
        if (--cnt[x] == 0)
        {
            missing.insert(x);
            cnt.erase(x);
        }
    }

    int query()
    {
        return *missing.begin();
    }
};

### 离线莫队mex
```cpp
struct Mex
{
    vector<int> arr, cnt, bcnt, ans; // cnt[x] = 值x在当前区间中的出现次数，bcnt[i] = 第i个值域块中缺失的数字个数（cnt=0的数字个数）
    vector<tuple<int, int, int>> qs;
    int n, block, val_block;

    Mex(const vector<int> &a) : arr(a)//初始数据a(0-based)
    {
        n = arr.size();
        int max_val = n + 2;
        val_block = sqrt(max_val) + 1;
        cnt.assign(max_val, 0);
        bcnt.assign((max_val + val_block - 1) / val_block, 0); // 初始化值域分块数组：共有 ceil(max_val / val_block) 个块

        // 初始化缺失计数
        for (int i = 0; i < bcnt.size(); i++)
        {
            int start = i * val_block;
            int end = min(max_val, start + val_block);
            bcnt[i] = end - start;
        }

        block = sqrt(n) + 1;
    }

    void add_q(int l, int r)
    {
        qs.emplace_back(l, r, qs.size());
    }

    void add(int idx)
    {
        int x = arr[idx];
        if (x < cnt.size() && ++cnt[x] == 1)
            bcnt[x / val_block]--;
    }

    void del(int idx)
    {
        int x = arr[idx];
        if (x < cnt.size() && --cnt[x] == 0)
            bcnt[x / val_block]++;
    }

    int mex()
    {
        int b = 0;
        while (b < bcnt.size() && bcnt[b] == 0)
            b++;
        if (b >= bcnt.size())
            return cnt.size();
        int start = b * val_block;
        int end = min((int)cnt.size(), start + val_block);
        for (int i = start; i < end; i++)
            if (cnt[i] == 0)
                return i;
        return cnt.size();
    }

    vector<int> solve()
    {
        if (qs.empty())
            return {};

        sort(qs.begin(), qs.end(), [&](const auto &a, const auto &b)
             {
            auto [l1, r1, i1] = a;
            auto [l2, r2, i2] = b;
            int b1 = l1 / block, b2 = l2 / block;
            if (b1 != b2) return b1 < b2;
            return (b1 & 1) ? (r1 > r2) : (r1 < r2); });//奇偶优化莫队。降低常数
            
        ans.resize(qs.size());
        int l = 0, r = -1;
        
        for (auto [ql, qr, idx] : qs)
        {
            while (l > ql)
                add(--l);
            while (r < qr)
                add(++r);
            while (l < ql)
                del(l++);
            while (r > qr)
                del(r--);
            ans[idx] = mex();
        }
        return ans;
    }
};
```
